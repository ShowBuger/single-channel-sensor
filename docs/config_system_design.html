1. 现有驱动 Bug

⚠
所有 MLX90393_set*() 直接覆写整个寄存器，不先读取现有值。begin() 中先调 setOverSampling(3)（写 REG 0x02 = 0x0003），再调 setDigitalFiltering(0)（写 REG 0x02 = 0x0000）→ OSR 被清零，实际以 OSR=0 运行，而非预期的 OSR=3。

OSR [1:0]

2. 修复：read-modify-write + configureAll()

2a. 通用 RMW helper（适用于初始化后的运行时调参）

/* mlx90393.c — static helper，所有 set*() 改用此函数 */
static uint8_t rmw_reg(MLX90393_Handle *h, uint8_t id,
                       uint8_t reg, uint16_t mask, uint16_t value)
{
    uint16_t cur;
    uint8_t st = MLX90393_readRegister(h, id, reg, &cur);
    if (error_process(st) != MLX90393_ERROR_NONE) return st;
    return MLX90393_writeRegister(h, id, reg, (cur & ~mask) | (value & mask));
}
/* 各 set*() 改为一行 */
uint8_t MLX90393_setGainSel(...)        { return rmw_reg(h,id,GAIN_SEL_REG,       GAIN_SEL_MASK,       gain<<GAIN_SEL_SHIFT); }
uint8_t MLX90393_setOverSampling(...)   { return rmw_reg(h,id,OSR_REG,             OSR_MASK,            osr<<OSR_SHIFT); }
uint8_t MLX90393_setDigitalFiltering(...){ return rmw_reg(h,id,DIG_FLT_REG,         DIG_FLT_MASK,        flt<<DIG_FLT_SHIFT); }
uint8_t MLX90393_setTemperatureComp(...) { return rmw_reg(h,id,TCMP_EN_REG,          TCMP_EN_MASK,        (en?1:0)<<TCMP_EN_SHIFT); }
uint8_t MLX90393_setBurstDataRate(...)   { return rmw_reg(h,id,BURST_DATA_RATE_REG,  BURST_DATA_RATE_MASK,(rate&0x3F)<<BURST_DATA_RATE_SHIFT); }

2b. MLX90393_configureAll() — 3 次写入，覆盖全部可配置字段

在 reset 后寄存器状态已知为0，直接写三个寄存器，无需 RMW。这是 begin() 调用的路径。

/* mlx90393.h — 新增 */
typedef struct {
    uint8_t gain_sel;   /* CONF1[6:4]  0–7,  7=×1 最低增益 */
    uint8_t res_x, res_y, res_z;  /* CONF3[10:5] 0–3 */
    uint8_t osr;        /* CONF3[1:0]  0–3, 3=×8 最慢最精 */
    uint8_t dig_filt;   /* CONF3[4:2]  0–7 */
    uint8_t tcmp_en;    /* CONF2[10]   0/1 */
    uint8_t burst_rate; /* CONF2[5:0]  0–63 */
} MLX90393_Config;
#define MLX90393_DEFAULT_CONFIG  { 7, 0,0,0, 3, 0, 0, 0 }
uint8_t MLX90393_configureAll(MLX90393_Handle *handle, uint8_t id, const MLX90393_Config *cfg);

/* mlx90393.c — 实现（reset 后调用） */
uint8_t MLX90393_configureAll(MLX90393_Handle *h, uint8_t id, const MLX90393_Config *cfg)
{
    uint8_t  st;
    uint16_t r;
    /* REG 0x00: GAIN_SEL[6:4] */
    r  = (uint16_t)((cfg->gain_sel & 0x07) << GAIN_SEL_SHIFT);
    st = MLX90393_writeRegister(h, id, 0x00, r);
    if (error_process(st)) return st;
    /* REG 0x01: TCMP_EN[10] | BURST_DATA_RATE[5:0] */
    r  = (uint16_t)((cfg->tcmp_en?1:0) << TCMP_EN_SHIFT)
       | (uint16_t)((cfg->burst_rate&0x3F) << BURST_DATA_RATE_SHIFT);
    st = MLX90393_writeRegister(h, id, 0x01, r);
    if (error_process(st)) return st;
    /* REG 0x02: RES_XYZ[10:5] | DIG_FILT[4:2] | OSR[1:0] */
    r  = (uint16_t)((((cfg->res_z&3)<<4)|((cfg->res_y&3)<<2)|(cfg->res_x&3)) << RES_XYZ_SHIFT)
       | (uint16_t)((cfg->dig_filt&7) << DIG_FLT_SHIFT)
       | (uint16_t)((cfg->osr&3) << OSR_SHIFT);
    return MLX90393_writeRegister(h, id, 0x02, r);
}
/* begin() 中替换两个 set*() 调用：*/
MLX90393_Config cfg = MLX90393_DEFAULT_CONFIG;
status = MLX90393_configureAll(handle, id, &cfg);

3. 扩展 Config_TypeDef（config_manager.h）

typedef struct {
    /* ── 算法标志 ── */
    uint8_t  FLAG_RES;                          /* 位掩码，各算法开关 */
    /* ── 传感器硬件 ── */
    uint8_t  gain_sel;                           /* 0–7,  默认 7         */
    uint8_t  osr;                                 /* 0–3,  默认 3         */
    uint8_t  dig_filt;                           /* 0–7,  默认 0         */
    uint8_t  res_x, res_y, res_z;               /* 0–3 各轴，默认 0    */
    uint8_t  tcmp_en;                            /* 0/1, 默认 0          */
    /* ── 映射系数 ── */
    float    X_MAPPING_VALUE;                    /* 默认  0.100          */
    float    Y_MAPPING_VALUE;                    /* 默认  0.100          */
    float    Z_MAPPING_VALUE;                    /* 默认 -0.100          */
    /* ── 算法调节 ── */
    float    dead_value;                         /* 死区阈值 µT，默认 8.5  */
    float    spike_multiplier;                   /* 尖峰σ倍数，默认 20.0 */
    float    stability_threshold;               /* 稳定性 µT，默认 0.1  */
    uint32_t checksum;
} Config_TypeDef;  /* ≈ 48 bytes，1 KB Flash 页绰绰有余 */
ℹ
SLIDING_WINDOW_SIZE 是编译期数组大小，不放入 Flash。通过 FLAG_RES bit 2 控制是否启用滑动窗口；窗口始终保持最大容量（5），运行时通过开关决定是否使用。

6. PC 工具参考实现（Python）

# tools/apply_config.py — 上位机配置下发脚本
import json, serial, time, sys

def apply_config(port, cfg_path):
    with open(cfg_path) as f:  cfg = json.load(f)
    with serial.Serial(port, 115200, timeout=1) as ser:
        cmds = []
        s = cfg["sensor"]
        cmds += [f"ML SET sensor.gain {s['gain_sel']}",
                 f"ML SET sensor.osr {s['osr']}",
                 f"ML SET sensor.filt {s['dig_filt']}",
                 f"ML SET sensor.resx {s['res_x']}",
                 f"ML SET sensor.resy {s['res_y']}",
                 f"ML SET sensor.resz {s['res_z']}",
                 f"ML SET sensor.tcmp {s['tcmp_en']}"]
        a = cfg["algorithm"]
        cmds += [f"ML SET algo.flag {a['flag_res']}",
                 f"ML SET algo.xmap {a['x_map']:.3f}",
                 f"ML SET algo.ymap {a['y_map']:.3f}",
                 f"ML SET algo.zmap {a['z_map']:.3f}",
                 f"ML SET algo.dead {a['dead_value']:.3f}",
                 f"ML SET algo.spike {a['spike_multiplier']:.3f}",
                 f"ML SET algo.stability {a['stability']:.3f}",
                 "ML SAVECFG", "ML APPLYCFG"]
        for cmd in cmds:
            ser.write((cmd + "\r\n").encode())
            time.sleep(0.05)
            resp = ser.readline().decode("utf-8", errors="replace").strip()
            print(f"{cmd:35s} → {resp}")

if __name__ == "__main__":
    apply_config(sys.argv[1], sys.argv[2])
    # Usage: python apply_config.py COM3 sensor_config.json
